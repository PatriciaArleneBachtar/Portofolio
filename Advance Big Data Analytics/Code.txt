%let n_iterations = 10000;
%let high_risk_customers = 617;
%let retention_mean = 0.55; 
%let retention_std = 0.10; 

data monte_carlo_results;
    call streaminit(42);
    do iteration = 1 to &n_iterations;
        retention_prob = rand('Normal', &retention_mean, &retention_std);
        if retention_prob < 0 then retention_prob = 0;
        if retention_prob > 1 then retention_prob = 1;
        retained_customers = round(&high_risk_customers * retention_prob);
        output;
    end;
run;

proc means data=monte_carlo_results
    n mean std min p5 p25 p50 p75 p95 max;
    var retained_customers retention_prob;
    title "Monte Carlo Simulation Results - Customer Retention";
run;

proc sql;
    create table scenario_summary as
    select
        min(retained_customers) as Worst_Case,
        round(mean(retained_customers),1) as Most_Likely,
        max(retained_customers) as Best_Case,
        round(mean(retention_prob)*100,2) as Avg_Retention_Rate_Pct,
        round(std(retained_customers),2) as Std_Dev
    from monte_carlo_results;
quit;

title "Scenario Summary";
proc print data=scenario_summary noobs;
run;

title "Distribution of Retained Customers (10,000 Iterations)";
proc sgplot data=monte_carlo_results;
    histogram retained_customers /
        fillattrs=(color=steelblue)
        transparency=0.3
        binwidth=5;
    density retained_customers / type=normal
        lineattrs=(color=red thickness=2);
    xaxis label="Number of Retained Customers";
    yaxis label="Frequency";
    refline 340 / axis=x lineattrs=(color=darkgreen pattern=dash)
        label="Most Likely";
run;
